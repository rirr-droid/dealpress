**2. PDF Export Function**
```typescript
// lib/exports/pdf.ts (NEW)
import { jsPDF } from 'jspdf'
import autoTable from 'jspdf-autotable'

export async function generateAnalyticsPDF(analytics: AnalyticsMetrics, orgName: string) {
  const doc = new jsPDF()

  // Header
  doc.setFontSize(20)
  doc.text('Approval Analytics Report', 20, 20)
  doc.setFontSize(12)
  doc.text(orgName, 20, 30)
  doc.text(`Generated on ${new Date().toLocaleDateString()}`, 20, 36)

  // Key Metrics
  doc.setFontSize(16)
  doc.text('Key Metrics', 20, 50)
  autoTable(doc, {
    startY: 55,
    head: [['Metric', 'Value']],
    body: [
      ['Total Requests', analytics.totalRequests],
      ['Approval Rate', `${analytics.approvalRate}%`],
      ['Avg Approval Time', `${analytics.averageApprovalTime} hours`],
      ['Pending Approvals', analytics.pendingRequests]
    ]
  })

  // Top Approvers
  doc.setFontSize(16)
  doc.text('Top Approvers', 20, doc.lastAutoTable.finalY + 20)
  autoTable(doc, {
    startY: doc.lastAutoTable.finalY + 25,
    head: [['Name', 'Approved', 'Avg Response Time']],
    body: analytics.topApprovers.map(a => [
      a.name,
      a.approved_count,
      `${a.avg_response_time.toFixed(1)}h`
    ])
  })

  // Footer
  doc.setFontSize(10)
  doc.setTextColor(150)
  doc.text('Generated by DealPress - dealpress.com', 20, 280)

  return doc.output('blob')
}
```

**3. CSV Export Function**
```typescript
// lib/exports/csv.ts (NEW)
export function generateAnalyticsCSV(analytics: AnalyticsMetrics) {
  // Convert analytics to CSV format
  const rows = [
    ['Metric', 'Value'],
    ['Total Requests', analytics.totalRequests],
    ['Approved', analytics.approvedRequests],
    ['Rejected', analytics.rejectedRequests],
    ['Pending', analytics.pendingRequests],
    ['Approval Rate', `${analytics.approvalRate}%`],
    ['Avg Approval Time (hours)', analytics.averageApprovalTime],
    [''],
    ['Top Approvers', ''],
    ['Name', 'Approved Count', 'Avg Response Time (hours)'],
    ...analytics.topApprovers.map(a => [a.name, a.approved_count, a.avg_response_time.toFixed(1)]),
    [''],
    ['Daily Trend', ''],
    ['Date', 'Total', 'Approved', 'Rejected'],
    ...analytics.requestTrend.map(t => [t.date, t.count, t.approved, t.rejected])
  ]

  return rows.map(row => row.join(',')).join('\n')
}
```

**4. Server Actions for Export**
```typescript
// app/actions/analytics.ts (NEW)
'use server'

export async function exportAnalyticsPDF() {
  const orgId = await getUserOrgId()
  const org = await getOrganization(orgId)
  const analytics = await getAnalytics(orgId, 30)

  const pdfBlob = await generateAnalyticsPDF(analytics, org.name)

  // Return base64 for download
  return Buffer.from(await pdfBlob.arrayBuffer()).toString('base64')
}

export async function exportAnalyticsCSV() {
  const orgId = await getUserOrgId()
  const analytics = await getAnalytics(orgId, 30)

  return generateAnalyticsCSV(analytics)
}

export async function generateAnalyticsShareLink() {
  const orgId = await getUserOrgId()
  const token = crypto.randomUUID()

  await supabase
    .from('organizations')
    .update({
      analytics_share_token: token,
      analytics_shared_at: new Date().toISOString()
    })
    .eq('id', orgId)

  return `${APP_URL}/analytics/share/${token}`
}
```

**7. Client-Side Download Handlers**
```typescript
// app/(dashboard)/analytics/page.tsx (CLIENT COMPONENT)
'use client'

async function handleExportPDF() {
  const base64PDF = await exportAnalyticsPDF()
  const blob = base64ToBlob(base64PDF, 'application/pdf')
  downloadBlob(blob, `analytics-${new Date().toISOString().split('T')[0]}.pdf`)
}

async function handleExportCSV() {
  const csvData = await exportAnalyticsCSV()
  const blob = new Blob([csvData], { type: 'text/csv' })
  downloadBlob(blob, `analytics-${new Date().toISOString().split('T')[0]}.csv`)
}

async function handleGenerateShareLink() {
  const link = await generateAnalyticsShareLink()
  // Show modal with link + copy button
  setShareLink(link)
}
```

## Security Considerations

- ✅ Public analytics page shows aggregated metrics only (no individual request details)
- ✅ Approver names optional in public view (privacy setting)
- ✅ Share token is UUID (unguessable)
- ✅ Admin can revoke share link anytime
- ⚠️ Rate limit PDF generation to prevent abuse (5 exports/hour/user)
- ⚠️ Rate limit public analytics route (100 views/hour/IP)

---

## Edge Cases

1. **Share link is revoked after sharing**
   - Show: "Analytics link has been revoked"

2. **Organization has no analytics data yet**
   - Show: "No data to display" with instructions to create requests

3. **PDF export fails (too much data)**
   - Show error: "Unable to generate PDF. Try exporting as CSV instead."
   - Limit PDF to last 30 days

4. **User generates multiple share links**
   - New link revokes old link (only one active at a time)

5. **CSV file is too large**
   - Paginate data or limit to last 90 days
